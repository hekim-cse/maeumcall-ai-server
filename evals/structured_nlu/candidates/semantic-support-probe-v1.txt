"""Probe whether a separate local model call can judge field meaning, development only."""

from __future__ import annotations

import argparse
import hashlib
import json
import time
import urllib.request
from datetime import UTC, datetime
from pathlib import Path

from evals.structured_nlu.semantic_support import CLAIM_SOURCE, VERDICTS, load_claims
from scripts.probe_gguf_field_boundary import inspect_runtime

ROOT = Path(__file__).resolve().parents[1]
SETTINGS = {
    "max_tokens": 100,
    "temperature": 0,
    "repeat_penalty": 1.05,
    "stop": ["<|eot_id|>"],
    "response_format": {"type": "json_object"},
}
INSTRUCTION = (
    "사용자 발화가 업무 필드의 후보 값을 의미상 뒷받침하는지 판단하세요. "
    "인용문이 원문에 있다는 사실만으로 supported로 판단하지 마세요. "
    "명시적으로 지지하면 supported, 반대하면 contradicted, "
    "모호하거나 그 값까지 확정할 수 없으면 insufficient입니다. "
    '정확히 {"verdict": "supported|contradicted|insufficient"} JSON만 출력하세요.'
)


def _sha256(path: Path) -> str:
    with path.open("rb") as stream:
        return hashlib.file_digest(stream, "sha256").hexdigest()


def judge_claim(claim: dict, *, endpoint: str, timeout: int = 180) -> dict:
    """Save only a bounded verdict, timing and token counts, never prompt/completion text."""
    payload = {
        "model": "kanana-1.5-2.1b-instruct-q5",
        "messages": [
            {"role": "system", "content": INSTRUCTION},
            {
                "role": "user",
                "content": json.dumps(
                    {
                        "utterance": claim["utterance"],
                        "field": claim["field"],
                        "candidate_value": claim["value"],
                        "quote": claim["quote"],
                    },
                    ensure_ascii=False,
                ),
            },
        ],
    } | SETTINGS
    request = urllib.request.Request(
        endpoint,
        data=json.dumps(payload, ensure_ascii=False).encode(),
        headers={"Content-Type": "application/json"},
        method="POST",
    )
    started = time.perf_counter()
    with urllib.request.urlopen(request, timeout=timeout) as response:
        body = json.load(response)
    completion = body["choices"][0]
    try:
        parsed = json.loads(completion["message"]["content"])
        verdict = parsed.get("verdict") if isinstance(parsed, dict) else None
        if not isinstance(verdict, str) or verdict not in VERDICTS:
            verdict = None
    except json.JSONDecodeError:
        verdict = None
    return {
        "claim_id": claim["id"],
        "expected": claim["expected"],
        "quote_present": True,
        "verdict": verdict,
        "correct": verdict == claim["expected"],
        "elapsed_seconds": round(time.perf_counter() - started, 3),
        "prompt_tokens": body.get("usage", {}).get("prompt_tokens"),
        "completion_tokens": body.get("usage", {}).get("completion_tokens"),
        "finish_reason": completion.get("finish_reason"),
    }


def main() -> None:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--container-name", required=True)
    parser.add_argument("--output", type=Path, required=True)
    args = parser.parse_args()
    if args.output.exists():
        parser.error("output already exists")
    claims = load_claims()
    runtime = inspect_runtime(args.container_name)
    observations = []
    try:
        for claim in claims:
            result = judge_claim(claim, endpoint=runtime["endpoint"])
            observations.append(result)
            print(f"{claim['id']} verdict={result['verdict']}", flush=True)
    finally:
        report = {
            "kind": "exploratory_q5_semantic_support_probe_v1",
            "not_official_quality_score": True,
            "complete": len(observations) == len(claims),
            "created_at_utc": datetime.now(UTC).isoformat(),
            "source_sha256": {
                name: _sha256(ROOT / name)
                for name in (
                    "evals/structured_nlu/semantic_support.py",
                    "scripts/probe_gguf_semantic_support.py",
                    "scripts/probe_gguf_field_boundary.py",
                )
            },
            "claim_source_sha256": _sha256(CLAIM_SOURCE),
            "model_sha256": runtime["model_sha256"],
            "server_image_digest": runtime["server_image_digest"],
            "runtime": {key: value for key, value in runtime.items() if key != "endpoint"},
            "request_settings": SETTINGS,
            "observations": observations,
        }
        args.output.write_text(json.dumps(report, ensure_ascii=False, indent=2) + "\n")


if __name__ == "__main__":
    main()
